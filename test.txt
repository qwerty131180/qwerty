accounts = [101, 102, 103, 104, 105, 106, 107, 108, 109, 110]

search_acc = int(input("Enter account ID to search: "))

def linear_search(accounts, search_acc):
    for i in range(len(accounts)):
        if accounts[i] == search_acc:
            return i
    return -1

def binary_search(accounts, search_acc):
    st = 0
    end = len(accounts) - 1

    while st <= end:
        mid = (st + end) // 2

        if accounts[mid] == search_acc:
            return mid
        elif search_acc < accounts[mid]:
            end = mid - 1
        else:
            st = mid + 1

    return -1

print("\nLINEAR SEARCH")
index = linear_search(accounts, search_acc)

if index != -1:
    print("Account found at index", index)
else:
    print("Account not found")

print("\nBINARY SEARCH")
index = binary_search(accounts, search_acc)

if index != -1:
    print("Account found at index", index)
else:
    print("Account not found")

# Experiment 2: Employee Salary Sorting (Selection Sort and Bubble Sort)
n = int(input("Enter number of employees: "))

salary = []
for i in range(n):
    s = float(input("Enter salary of employee " + str(i + 1) + ": "))
    salary.append(s)


salary1 = salary.copy()   # will be sorted using Selection Sort
salary2 = salary.copy()   # will be sorted using Bubble Sort

#Selection Sort 
for i in range(n - 1):
    small = i                     
    for j in range(i + 1, n):
        if salary1[j] < salary1[small]:
            small = j              
    
    temp = salary1[i]
    salary1[i] = salary1[small]
    salary1[small] = temp

#Bubble Sort
for i in range(n - 1):
    for j in range(n - 1 - i):
        if salary2[j] > salary2[j + 1]:
        
            temp = salary2[j]
            salary2[j] = salary2[j + 1]
            salary2[j + 1] = temp


print("\nSalaries after Selection Sort:", salary1)
print("Salaries after Bubble Sort:", salary2)


print("\nTop 5 highest salaries:")
count = 0
for i in range(n - 1, -1, -1):
    if count == 5:
        break
    print(count + 1, ".", salary1[i])
    count = count + 1

# Experiment 3: Undo/Redo System Using Stack
undo_stack=[]
redo_stack=[]
document=""

def make_change():
    global document
    undo_stack.append(document)

    new_text=input("Enter your change:")
    document+=new_text

    redo_stack.clear()
    print("Changes made to the document.")

def undo():
    global document
    if len(undo_stack)==0:
        print("No change to undo")

    else:
        redo_stack.append(document)
        document=undo_stack.pop()
        print("Undo successful.")

def redo():
    global document
    if len(redo_stack)==0:
        print("No change to redo")

    else:
        undo_stack.append(document)
        document=redo_stack.pop()
        print("Redo successful.")

while True:
    print("\nCurrent document:",document)
    print("1. Make a change")
    print("2. Undo")
    print("3. Redo")
    print("4. Exit")

    choice=int(input("Enter your choice:"))

    if choice==1:
        make_change()
    elif choice==2:
        undo()  
    elif choice==3:
        redo()
    elif choice==4:
        print("Exiting the program.")
        break
    else:
        print("Invalid choice. Please try again.")

# Experiment 4: Student Record Management Using Singly Linked List
class Node:
    def __init__(self, roll, name, marks):
        self.roll = roll
        self.name = name
        self.marks = marks
        self.next = None
 
 
class StudentList:
    def __init__(self):
        self.head = None
 
    def add(self, roll, name, marks):
        new_node = Node(roll, name, marks)
        if self.head is None:
            self.head = new_node
            return
        temp = self.head
        while temp.next is not None:
            temp = temp.next
        temp.next = new_node
 
    def search(self, roll):
        temp = self.head
        while temp is not None:
            if temp.roll == roll:
                return temp
            temp = temp.next
        return None
 
    def delete(self, roll):
        temp = self.head
        previous = None
        while temp is not None and temp.roll != roll:
            previous = temp
            temp = temp.next
        if temp is None:
            print("Student Not Found")
            return
        if previous is None:            # deleting the first node
            self.head = temp.next
        else:
            previous.next = temp.next
 
    def update(self, roll, new_name, new_marks):
        student = self.search(roll)
        if student is None:
            print("Student Not Found")
        else:
            student.name = new_name
            student.marks = new_marks
 
    def sort(self, descending=False):
        # Bubble sort on the linked list (swap the data, not the nodes)
        swapped = True
        while swapped:
            swapped = False
            temp = self.head
            while temp is not None and temp.next is not None:
                if descending:
                    wrong_order = temp.marks < temp.next.marks
                else:
                    wrong_order = temp.marks > temp.next.marks
                if wrong_order:
                    temp.roll, temp.next.roll = temp.next.roll, temp.roll
                    temp.name, temp.next.name = temp.next.name, temp.name
                    temp.marks, temp.next.marks = temp.next.marks, temp.marks
                    swapped = True
                temp = temp.next
 
    def display(self):
        temp = self.head
        while temp is not None:
            print(" [" + str(temp.roll) + ", " + temp.name + ", " + str(temp.marks) + "]", end=" ->")
            temp = temp.next
        print(" NULL")
 
 
students = StudentList()
students.add(101, "Amit", 85)
students.add(102, "Ravi", 72)
students.add(103, "Neha", 91)
students.add(104, "Sara", 64)

    index=key%size

    hash_table[index].append((key,value))
    print("Key-value pair inserted at index:", index)
    return

    index=key%size

    for k,v in hash_table[index]:
        if k==key:
            print("Key found at index:", index, "with value:", v)

# Experiment 8: Binary Search Tree Operations

class Node:
    def __init__(self, data):
        self.data = data
        self.left = None
        self.right = None


def insert(root, data):
    if root is None:
        return Node(data)

    if data < root.data:
        root.left = insert(root.left, data)

    elif data > root.data:
        root.right = insert(root.right, data)

    else:
        print("Duplicate value not allowed")

    return root


def search(root, key):
    if root is None:
        return False

    if root.data == key:
        return True

    elif key < root.data:
        return search(root.left, key)

    else:
        return search(root.right, key)


def inorder(root):
    if root is not None:
        inorder(root.left)
        print(root.data, end=" ")
        inorder(root.right)


def delete(root, key):
    if root is None:
        return None

    if key < root.data:
        root.left = delete(root.left, key)

    elif key > root.data:
        root.right = delete(root.right, key)

    else:
        # Case 1: No child
        if root.left is None and root.right is None:
            return None

        # Case 2: One child
        elif root.left is None:
            return root.right

        elif root.right is None:
            return root.left

        # Case 3: Two children
        else:
            temp = root.right

            while temp.left is not None:
                temp = temp.left

            root.data = temp.data
            root.right = delete(root.right, temp.data)

    return root


root = None

            print("Value not found")
